# --- Ядро: нейросеть и работа со звуком -------------------------------------

# torch — фреймворк для нейросетей (обучение и вычисления)
torch
# torchaudio — обработка звука средствами PyTorch (совместима с Demucs)
torchaudio
# demucs — сама архитектура разделения музыки (лицензия MIT)
demucs
# numpy — работа с массивами чисел
numpy
# librosa — чтение и анализ аудиофайлов, подсчёт метрик качества
librosa
# soundfile — чтение mp3/flac/ogg/wav и запись 32-битного float WAV.
#              Нужен потому, что в этой версии torchaudio.load/save
#              требует пакет torchcodec, которого нет. Со soundfile
#              конвертер работает без ffmpeg.
soundfile

# --- Веб-интерфейс (API) ----------------------------------------------------

# fastapi — сам HTTP-сервис (Swagger-документация идёт в комплекте)
fastapi
# uvicorn — сервер, который запускает FastAPI
uvicorn
# pydantic — проверка и описание данных в запросах/ответах
pydantic
# python-multipart — ОБЯЗАТЕЛЬНО для загрузки файлов через FastAPI.
#                    Без него ручки приёма файлов не работают (ошибка 422).
python-multipart
# aiofiles — асинхронная работа с файлами (для больших загрузок)
aiofiles

# --- Telegram-бот ------------------------------------------------------------

# python-telegram-bot — библиотека для Telegram-ботов
python-telegram-bot

# ---------------------------------------------------------------------------
# ЕСЛИ ЧТО-ТО ПОШЛО НЕ ТАК
# ---------------------------------------------------------------------------
#
# 1) pip ругается, что не находит torch
#    Ставь torch отдельно с официальной страницы под свою систему
#    (https://pytorch.org/get-started/locally/), затем ставь остальное:
#        pip install -r requirements.txt
#
# 2) Ставится очень долго / съедает много места
#    torch весит ~2 ГБ, это нормально. Если нужна только CPU-версия
#    (около 200 МБ), ставь так:
#        pip install torch torchaudio --index-url https://download.pytorch.org/whl/cpu
#    и закомментируй строки torch/torchaudio выше.
#
# 3) Ошибка про ffmpeg (бывает при чтении mp3)
#    Сейчас ffmpeg НЕ обязателен: mp3, flac, ogg и wav читает soundfile.
#    Если всё же ругается, что ffmpeg не найден, поставь его и добавь
#    папку в PATH:
#        https://ffmpeg.org/download.html
#
# 4) Ошибка "TorchCodec is required" (от torchaudio.load / torchaudio.save)
#    Это не поломка, а особенность версии: torchaudio 2.11 требует
#    пакет torchcodec. Наш код это учитывает и читает звук через
#    soundfile, поэтому на конвертер и data_prep не влияет.
#
# 4) Несовместимость версий
#    Посмотреть, что уже стоит:      pip list
#    Обновить нужную библиотеку:    pip install --upgrade torch
#    Начать с чистого листа:         pip uninstall torch && pip install torch
